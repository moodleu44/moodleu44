#!/bin/bash
DATE=$(date +%Y-%m-%d)
BACKUP_DIR="/backup"
# Dump database Moodle dan kompres ke tar
mysqldump -u root -p'PASSWORD_ROOT_MARIADB' db_moodle > $BACKUP_DIR/db_moodle_$DATE.sql
tar -cvf $BACKUP_DIR/backup_db_$DATE.tar -C $BACKUP_DIR db_moodle_$DATE.sql
# Hapus file dump raw sql, sisakan file .tar
rm $BACKUP_DIR/db_moodle_$DATE.sql
# Hapus file backup .tar di folder /backup yang usianya lebih dari 10 hari
find $BACKUP_DIR -type f -name "*.tar" -mtime +10 -delete
chmod +x /usr/local/bin/backup_db.sh
crontab -e
0 1 * * * /usr/local/bin/backup_db.sh
